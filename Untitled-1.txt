    ];

    // --------------------------
    // State & elemen DOM

    let vocab = [];
    let score = 0, correct=0, wrong=0, qIndex=0;

    const modeSelect = el('modeSelect'), typeInputWrap = el('typeInput'), typedAnswer = el('typedAnswer'), submitTyped = el('submitTyped');
    const hintBtn = el('hintBtn'), showAnswerBtn = el('showAnswerBtn'), skipBtn = el('skipBtn');

    // --------------------------
    // LocalStorage helpers
    // --------------------------
    const STORAGE_KEY = 'game_en_vocab_v1';
    const STATS_KEY = 'game_en_stats_v1';

    function loadData(){
      const raw = localStorage.getItem(STORAGE_KEY);
      if(raw) {
        try{ vocab = JSON.parse(raw); if(!Array.isArray(vocab) || vocab.length===0) throw 'bad'; }
        catch(e){ vocab = DEFAULT_VOCAB.slice(); saveData(); }
      } else { vocab = DEFAULT_VOCAB.slice(); saveData(); }

      const statsRaw = localStorage.getItem(STATS_KEY);
      if(statsRaw){ try{ const s = JSON.parse(statsRaw); score=s.score||0; correct=s.correct||0; wrong=s.wrong||0; qIndex=s.qIndex||0;}catch(e){} }
    }

    function saveData(){ localStorage.setItem(STORAGE_KEY, JSON.stringify(vocab)); }
    function saveStats(){ localStorage.setItem(STATS_KEY, JSON.stringify({score,correct,wrong,qIndex})); }

    // --------------------------
    // UI: tabel kosakata
    // --------------------------
    function renderVocabTable(){
      const tbody = document.querySelector('#vocabTable tbody'); tbody.innerHTML='';
      vocab.forEach((w,i)=>{
        const tr = document.createElement('tr');
        tr.innerHTML = `<td>${i+1}</td><td>${escapeHtml(w.word)}</td><td>${escapeHtml(w.meaning)}</td><td>${escapeHtml(w.example||'')}</td><td>
          <button data-id="${w.id}" class="ghost edit">Edit</button>
          <button data-id="${w.id}" class="ghost del">Hapus</button>
        </td>`;
        tbody.appendChild(tr);
      });

      // attach handlers
      document.querySelectorAll('.edit').forEach(b=>b.onclick=e=>{
        const id = +e.target.dataset.id; const w = vocab.find(x=>x.id===id); if(!w) return;
        el('wordInput').value = w.word; el('meaningInput').value = w.meaning; el('exampleInput').value = w.example||'';
        // remove old entry
        vocab = vocab.filter(x=>x.id!==id); saveData(); renderVocabTable();
      });
      document.querySelectorAll('.del').forEach(b=>b.onclick=e=>{
        const id = +e.target.dataset.id; if(!confirm('Hapus kata ini?')) return; vocab = vocab.filter(x=>x.id!==id); saveData(); renderVocabTable();
      });
    }

    // --------------------------
    // Game logic
    // --------------------------
    function startGame(){
      if(vocab.length<2) { alert('Tambahkan lebih banyak kata (minimal 2).'); return; }
      qIndex = 0; score = 0; correct = 0; wrong = 0; updateStatsUI(); nextQuestion(); saveStats();
    }

    function nextQuestion(){
      qIndex++;
      qIndexEl.textContent = qIndex;
      const mode = modeSelect.value;
      currentQuestion = makeQuestion(mode);
      showQuestion(currentQuestion, mode);
      saveStats();
    }

    function makeQuestion(mode){
      // pilih 1 kata target
      const target = randomChoice(vocab);
      if(mode==='mc'){
        // buat 3 distractor
        const others = vocab.filter(v=>v.word!==target.word);
        const opts = [target.word];
        shuffleArray(others);
        for(let i=0;i<Math.min(3,others.length);i++) opts.push(others[i].word);
        shuffleArray(opts);
        return {target,options:opts};
      } else {
        return {target};
      }
    }
